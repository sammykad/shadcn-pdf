## Features

- `PDFDocument` registers Geist fonts automatically — use it instead of a raw `<Document>`.
- `PDFPage` applies the active font family, background, and default padding.
- `PDFHeader` / `PDFFooter` with an optional automatic page number.
- Zero-config — no provider or theme setup needed.

## Usage

```tsx
import { PDFDocument, PDFPage, PDFHeader, PDFFooter } from "@/components/pdf/document";
import { PDFHeading, PDFTextBlock } from "@/components/pdf/typography";
import { View } from "@react-pdf/renderer";

<PDFDocument title="My Document" author="Me">
  <PDFPage>
    <PDFHeader>
      <View style={{ flexDirection: "column", gap: 4 }}>
        <PDFHeading level={2}>Header</PDFHeading>
        <PDFTextBlock variant="small" color="#737373">Subtitle</PDFTextBlock>
      </View>
    </PDFHeader>
    <PDFTextBlock>Body content here.</PDFTextBlock>
    <PDFFooter page={1} right="shadcn-pdf" />
  </PDFPage>
</PDFDocument>
```
