## Features

- A bordered, rounded container for grouping PDF content.
- Zero-config — no provider needed.
- Supports `className` prop for Tailwind-style overrides.

<Callout type="info">
  `card` is a PDF component rendered with @react-pdf/renderer, not a DOM
  element. It is meant to live inside a `PDFDocument`/`PDFPage`.
</Callout>

## Usage

```tsx
import { PDFDocument, PDFPage } from "@/components/pdf/document";
import {
  PDFCard,
  PDFCardHeader,
  PDFCardTitle,
  PDFCardDescription,
  PDFCardContent,
  PDFCardFooter,
} from "@/components/pdf/card";
import { renderToBuffer } from "@react-pdf/renderer";

function InvoiceCard() {
  return (
    <PDFDocument title="Invoice" author="Me">
      <PDFPage>
        <PDFCard>
          <PDFCardHeader>
            <PDFCardTitle>Invoice #001</PDFCardTitle>
            <PDFCardDescription>Acme Corp</PDFCardDescription>
          </PDFCardHeader>
          <PDFCardContent>Content here</PDFCardContent>
          <PDFCardFooter>Footer here</PDFCardFooter>
        </PDFCard>
      </PDFPage>
    </PDFDocument>
  );
}

// Render to buffer
const buffer = await renderToBuffer(<InvoiceCard />);
```
