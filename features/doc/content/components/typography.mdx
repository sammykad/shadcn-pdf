## Features

- `PDFHeading` — semantic heading levels (1–4) from the theme type scale.
- `PDFTextBlock` — body, small, and mono text variants.
- Supports color and alignment overrides per instance.
- Supports `className` prop for Tailwind-style overrides.

## Usage

```tsx
import { PDFDocument, PDFPage } from "@/components/pdf/document";
import { PDFHeading, PDFTextBlock } from "@/components/pdf/typography";
import { renderToBuffer } from "@react-pdf/renderer";

function MyDocument() {
  return (
    <PDFDocument title="My Document" author="Me">
      <PDFPage>
        <PDFHeading level={2}>Section heading</PDFHeading>
        <PDFTextBlock variant="small" color="#737373">Supporting detail</PDFTextBlock>
      </PDFPage>
    </PDFDocument>
  );
}

// Render to buffer
const buffer = await renderToBuffer(<MyDocument />);
```

## With className

```tsx
<PDFHeading level={1} className="text-4xl font-bold">Large title</PDFHeading>
<PDFTextBlock className="text-muted-foreground">Muted text</PDFTextBlock>
```

## API reference

### PDFHeading

| Prop | Type |
| --- | --- |
| children | `ReactNode` |
| level? | `1 \| 2 \| 3 \| 4` |
| color? | `string` |
| align? | `"left" \| "center" \| "right"` |
| className? | `string` |
| style? | `Style` |

### PDFTextBlock

| Prop | Type |
| --- | --- |
| children | `ReactNode` |
| variant? | `"body" \| "small" \| "mono"` |
| color? | `string` |
| align? | `"left" \| "center" \| "right"` |
| className? | `string` |
| style? | `Style` |
