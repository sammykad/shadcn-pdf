  return (

      <PDFPage>
        <PDFContainer className="flex-col gap-3">
          <PDFText>Above the divider</PDFText>

        </PDFContainer>
      </PDFPage>

  );
}

## Features

- A single horizontal line using the theme border color.
- Full-width by default, with optional `style` overrides.
- Supports `className` prop for Tailwind-style overrides.

## Usage

```tsx
import { PDFDocument, PDFPage } from "@/components/pdf/document";
import { PDFDivider } from "@/components/pdf/divider";
import { PDFText } from "@/components/pdf";
import { renderToBuffer } from "@react-pdf/renderer";

function MyDocument() {
  return (
    <PDFDocument title="My Document" author="Me">
      <PDFPage>
        <PDFText>Content above</PDFText>
        <PDFDivider />
        <PDFText>Content below</PDFText>
      </PDFPage>
    </PDFDocument>
  );
}

// Render to buffer
const buffer = await renderToBuffer(<MyDocument />);
```
