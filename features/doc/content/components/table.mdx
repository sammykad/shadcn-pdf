## Features

- Layout primitives for tabular data in PDF documents.
- Column widths via `flex` prop — no percentage hacks.
- Strings auto-wrap in `<Text>`, React nodes pass through.
- Supports `className` prop for Tailwind-style overrides.

<Callout type="info">
  `table` is a set of PDF components rendered with @react-pdf/renderer, not DOM
  elements. They are meant to live inside a `PDFDocument`/`PDFPage`.
</Callout>

## Usage

```tsx
import { PDFDocument, PDFPage } from "@/components/pdf/document";
import {
  PDFTable,
  PDFTableHeader,
  PDFTableBody,
  PDFTableRow,
  PDFTableHead,
  PDFTableCell,
  PDFTableFooter,
} from "@/components/pdf/table";
import { renderToBuffer } from "@react-pdf/renderer";

function InvoiceTable() {
  return (
    <PDFDocument title="Invoice" author="Me">
      <PDFPage>
        <PDFTable>
          <PDFTableHeader>
            <PDFTableHead flex={2}>Item</PDFTableHead>
            <PDFTableHead flex={1}>Qty</PDFTableHead>
            <PDFTableHead flex={1}>Price</PDFTableHead>
          </PDFTableHeader>
          <PDFTableBody>
            <PDFTableRow>
              <PDFTableCell flex={2}>Widget</PDFTableCell>
              <PDFTableCell flex={1}>2</PDFTableCell>
              <PDFTableCell flex={1}>$19.00</PDFTableCell>
            </PDFTableRow>
          </PDFTableBody>
        </PDFTable>
      </PDFPage>
    </PDFDocument>
  );
}

// Render to buffer
const buffer = await renderToBuffer(<InvoiceTable />);
```

## Column widths

Use `flex` prop to control column widths:

```tsx
<PDFTableHead flex={2}>Wide column</PDFTableHead>
<PDFTableHead flex={1}>Narrow column</PDFTableHead>
```
